df1<-flights %>%
  group_by(dest) %>%
  summarise(mean_arr_delay=mean(arr_delay, na.rm = TRUE),
            mean_dep_delay=mean(dep_delay, na.rm = TRUE)) %>%
  filter(dest %in% c("PDX", "LNK", "JAC", "SAT", "MIA"))

df2<-airports %>%
  filter(faa %in% c("PDX", "LNK", "JAC", "SAT", "MIA")) %>%
  select(dest="faa","name","lat","lon")

df <- left_join(df1, df2, by="dest")

ggplot(stat = "identity") + 
  geom_polygon(data=s, 
               aes(x=long, y=lat,group=group, fill=group)) + 
  geom_point(data = df,
             aes(x = lon, y = lat, group=dest),
             color = "red", size=2) +
  geom_text(data=df, aes(x=lon, y=lat, group=dest, label=paste("Arr_Del :",round(mean_arr_delay,2),"\n","Dep_del :",round(mean_dep_delay,2))), vjust=-0.25, size=4, color="red")+
  coord_map("polyconic")+
  labs(
    title = "U.S. airport flight delays",
    subtitle = "Mean arrival and departure delays in some states",
    x=NULL,
    y=NULL
  )+
  my_theme+
  theme(legend.position = "none",
        axis.text = element_blank(),
        axis.ticks = element_blank())

Visualization 2

This visualization was created using data from my project

state %>% 
  distinct(occ_title, .keep_all = TRUE) %>%
  filter(str_detect(occ_code,"29-|31-"),!(tot_emp=="**"), !(a_mean %in% c("*","#"))) %>%
  mutate(
    a_mean=as.numeric(a_mean),
    tot_emp=as.numeric(tot_emp)
  )%>%
  group_by(area_title) %>%
  summarise(mean=mean(a_mean)) %>%
  mutate(
    region = str_to_lower(area_title)
  ) %>%
  right_join(s, by="region") %>%
  ggplot(aes(x=long, y=lat, group=group, fill=mean))+
  geom_polygon(color="black")+
  coord_map("polyconic")+
  my_theme+
  theme(
    legend.position = c(0.95, 0.25),
    axis.text = element_blank(),
    axis.ticks = element_blank(),
    legend.title = element_text()
  )+
  labs(
    title = "Healthcare Occupation Mean Wage",
    subtitle = "in U.S. state",
    x=NULL,
    y=NULL
  )+
  guides(fill=guide_legend(title="Mean Wage"))
