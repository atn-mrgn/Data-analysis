df <- state %>%

  mutate(
    a_mean=as.numeric(a_mean),

Visualization 1

centers_in<-filter(centers,id %in% c("AL","AK","AZ","CA","MN"))
df %>% 
  group_by(area_title) %>%
  summarise(mean=mean(a_mean)) %>%
  left_join(mine_spdf, by=c("area_title"="id")) %>%
  ggplot() +
  geom_polygon(aes(fill = mean, x = long, y = lat, group = group) , size=0, alpha=0.9) +
  geom_text(data=centers_in, aes(x=x, y=y, label=id), size=5)+
  scale_fill_viridis_c(option = "C", labels=scales::dollar, na.value = "grey50")+
  theme_void(base_size = 18,
             base_family = "Poppins")+
  theme(
    legend.position = "right",
    legend.title = element_blank(),
    plot.title.position = "panel"
  )+
  labs(
    title = "Healthcare Occupation Mean Annual Wage",
    subtitle = "in U.S. state",
    x=NULL,
    y=NULL
  )+
  coord_map()

df %>%
